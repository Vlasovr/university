Настрока точек доступа

но скрины настройки будут предствалены
Настройка DNS:
alpine:~# echo "nameserver 1.1.1.1" > /etc/resolv.conf
alpine:~# echo "nameserver 8.8.8.8" >> /etc/resolv.conf
Запись автозагрузки VLAN-модуля:
alpine:~# echo 8021q >> /etc/modules
Декларативная настрока интерфейсов:
alpine:~# vi /etc/network/interfaces
Текст для вставки:
_________________________
auto lo
iface lo inet loopback

auto eth0
iface eth0 inet manual

auto eth0.100
iface eth0.100 inet static
    address 192.168.100.3 (100.4, 100.5)
    netmask 255.255.255.248
    gateway 192.168.100.1
__________________________

alpine:~# vi /etc/local.d/ap.start
Текст для вставки:
_________________________


modprobe 8021q

ip link add link eth0 name eth0.20 type vlan id 20
ip link set eth0.20 up

ip link add br-client type bridge
ip link set br-client up

ip link set eth0.20 master br-client
ip link set eth1 master br-client
ip link set eth1 up
__________________________
alpine:~# chmod +x /etc/local.d/ap.start
alpine:~# rc-update add local default
alpine:~# rc-service local restart
